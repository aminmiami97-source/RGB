local function updateScale()
    -- Adjusts UI size based on screen resolution (Mobile/Tablet vs PC)
    local viewport = workspace.CurrentCamera.ViewportSize
    if viewport.X < 800 then
        UIScale.Scale = 0.8 -- Mobile
    elseif viewport.X < 1200 then
        UIScale.Scale = 1.0 -- Tablet / Small PC
    else
        UIScale.Scale = 1.2 -- Large PC
    end
end
updateScale()
workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale)

-- Helper function to create round buttons
local function createControlButton(text, color)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 25, 0, 25)
    btn.BackgroundColor3 = color
    btn.BackgroundTransparency = 0.5
    btn.Text = text
    btn.TextColor3 = Theme.Text
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 14
    btn.AutoButtonColor = false
    btn.Parent = ControlContainer
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = btn
    
    -- Hover Animation
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0.2, TextColor3 = Theme.Background}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0.5, TextColor3 = Theme.Text}):Play()
    end)
    
    return btn
end

-- // TAB LOGIC & ANIMATIONS
local activeTab = MainTab

local function switchTab(tabToSelect, contentToShow, contentToHide)
    if activeTab == tabToSelect then return end
    
    -- Reset old tab
    TweenService:Create(activeTab, TweenInfo.new(0.2), {TextColor3 = Theme.SubText, BackgroundTransparency = 0.8}):Play()
    activeTab.UIStroke.Transparency = 1
    
    -- Hide old content smoothly
    TweenService:Create(contentToHide, TweenInfo.new(0.15), {BackgroundTransparency = 1}):Play()
    task.wait(0.15)
    contentToHide.Visible = false
    
    -- Set new tab
    activeTab = tabToSelect
    TweenService:Create(activeTab, TweenInfo.new(0.2), {TextColor3 = Theme.Text, BackgroundTransparency = 0.5}):Play()
    activeTab.UIStroke.Transparency = 0.8
    
    -- Show new content smoothly
    contentToShow.BackgroundTransparency = 1
    contentToShow.Visible = true
end

MainTab.MouseButton1Click:Connect(function() switchTab(MainTab, MainContent, CreditsContent) end)
CreditsTab.MouseButton1Click:Connect(function() switchTab(CreditsTab, CreditsContent, MainContent) end)

-- // WINDOW CONTROLS LOGIC
local isMinimized = false
local originalSize = MainFrame.Size

MinimizeBtn.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    if isMinimized then
        -- Minimize animation
        TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(0, 500, 0, 50)}):Play()
        -- Hide content
        for _, v in pairs(MainFrame:GetChildren()) do
            if v:IsA("Frame") and v.Name ~= "Header" then
                TweenService:Create(v, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
                v.Visible = false
            end
        end
    else
        -- Restore animation
        TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = originalSize}):Play()
        -- Show content
        for _, v in pairs(MainFrame:GetChildren()) do
            if v:IsA("Frame") and v.Name ~= "Header" then
                v.Visible = true
                if v.Name == "TabContainer" then
                    TweenService:Create(v, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
                elseif v.Name == "ContentContainer" then
                    TweenService:Create(v, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
                end
            end
        end
    end
end)

CloseBtn.MouseButton1Click:Connect(function()
    -- Close animation
    local closeTween = TweenService:Create(MainFrame, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
        Size = UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 1
    })
    closeTween:Play()
    closeTween.Completed:Wait()
    
    -- Clean instance (Destroys the GUI entirely)
    ScreenGui:Destroy()
end)

-- // DRAG SYSTEM (For PC)
local dragging = false
local dragInput, mousePos, framePos

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        mousePos = input.Position
        framePos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - mousePos
        MainFrame.Position = UDim2.new(framePos.X.Scale, framePos.X.Offset + delta.X, framePos.Y.Scale, framePos.Y.Offset + delta.Y)
    end
end)
